RETURN a.id as adr, a.title as decision, c.name as enables, c.importance as importance ORDER BY importance DESC;

// Q14: Calculate concept centrality (most connected)
MATCH (c:CONCEPT)-[r]-() 
RETURN c.name as concept, COUNT(r) as connectivity ORDER BY connectivity DESC;

// Q15: Find roles and their applicable patterns
MATCH (r:ROLE)-[:HAS]->(c:CONCEPT)<-[:IMPLEMENTS|USES]-(p:PATTERN)
RETURN DISTINCT r.name as role, p.name as pattern, c.name as shared_concept LIMIT 15;

// Q16: Show test coverage gaps (ADRs not covered by tests)
MATCH (a:ADR) 
WHERE NOT EXISTS((a)-[:EXERCISES]->(:TEST))
RETURN a.id as adr_id, a.title as adr_title;

// Q17: Complexity distribution of concepts
MATCH (c:CONCEPT) 
RETURN c.complexity as complexity, COUNT(c) as count, COLLECT(c.name) as concepts;

// Q18: Find alternative approaches (concepts with ALTERNATIVE_TO relationships)
MATCH (c1:CONCEPT)-[:ALTERNATIVE_TO]->(c2:CONCEPT)
RETURN c1.name as primary_approach, c2.name as alternative, c1.chapter as chapter ORDER BY chapter;

// Q19: Show full path from test to concept through ADR
MATCH path = (t:TEST)-[:EXERCISES]->(a:ADR)-[:IMPLEMENTS|APPLIES_TO]->(c:CONCEPT)
RETURN t.testId as test, a.id as adr, c.name as concept LIMIT 10;

// Q20: Calculate importance scores for comprehensive learning path
MATCH (c:CONCEPT) 
RETURN c.name as concept, c.importance as base_importance, c.chapter as chapter 
ORDER BY c.importance DESC, c.chapter;

// ADVANCED ANALYSIS QUERIES
// ========================================

// Q21: Find concept clusters (groups of closely connected concepts)
MATCH (c1:CONCEPT)-[r:FOUNDATION|ENABLES|SUPPORTS|COMPLEMENTS]-(c2:CONCEPT)
RETURN c1.name as concept1, TYPE(r) as relationship, c2.name as concept2, r.strength as strength
ORDER BY r.strength DESC;

// Q22: Role-based learning paths (what role should learn what concepts)
MATCH (r:ROLE)-[:HAS]->(c:CONCEPT)
WITH r, COUNT(c) as concept_count, COLLECT(c.name) as concepts
RETURN r.name as role, concept_count, concepts
ORDER BY concept_count DESC;

// Q23: Pattern distribution by category
MATCH (p:PATTERN)
RETURN p.category as pattern_category, COUNT(p) as count, COLLECT(p.name) as patterns
ORDER BY count DESC;

// Q24: Relationship strength distribution
MATCH ()-[r]->() 
RETURN TYPE(r) as relationship_type, COUNT(r) as count, 
       MIN(COALESCE(r.strength, 0)) as min_strength,
       AVG(COALESCE(r.strength, 0)) as avg_strength,
       MAX(COALESCE(r.strength, 0)) as max_strength
ORDER BY avg_strength DESC;

// Q25: Concept prerequisites (what should be learned first)
MATCH (c:CONCEPT)
OPTIONAL MATCH (c)<-[:FOUNDATION|ENABLES]-(prerequisite:CONCEPT)
RETURN c.name as concept, c.importance as importance, c.complexity as complexity,
       COLLECT(DISTINCT prerequisite.name) as prerequisites
ORDER BY c.importance DESC;
